cmake_minimum_required(VERSION 3.20)
project(osc_core LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(osc_core src/controller.c)
target_include_directories(osc_core PUBLIC include)
if(MSVC)
  target_compile_options(osc_core PRIVATE /W4 /WX)
else()
  target_compile_options(osc_core PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
enable_testing()
add_executable(test_controller ../../tests/test_controller.c)
target_link_libraries(test_controller PRIVATE osc_core)
add_test(NAME controller_lockout COMMAND test_controller)
